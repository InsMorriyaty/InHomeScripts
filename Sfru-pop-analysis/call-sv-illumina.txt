1. bam获取
bwa比对、picard去除重复序列

2. manta 
singularity exec ~/software/manta_v1.6.0.sif configManta.py --bam S1.sort.markdup.bam --referenceFasta target.fa --runDir S1
singularity exec ~/software/manta_v1.6.0.sif python S1/runWorkflow.py
python3 MantaFilter.py S1
在MantaFilterVcf文件夹中有所有个体的manta vcf， 如 S1.filter.vcf

3. delly
singularity exec ~/software/delly_v1.7.2.sif delly call -g target.fa -o S1.bcf S1.sort.markdup.bam  # S2 S3... Sn
singularity exec ~/software/delly_v1.7.2.sif delly merge -o all.sites.bcf input.bcf.txt
singularity exec ~/software/delly_v1.7.2.sif delly call -g target.fa -v all.sites.bcf -o S1.geno.bcf S1.sort.markdup.bam # S2 S3... Sn
bcftools view S1.geno.bcf -O v > S1.vcf
python3 DellyFilter.py S1

4. smoove
smoove call --name S1 --fasta target.fa  --genotype S1.sort.markdup.bam --outdir S1 # S2 S3... Sn
smoove merge -name merge -fasta target.fa output/*.vcf.gz #把所有结果放在一个文件夹里
smoove genotype -d -x -p 12 --name S1-joint --outdir results_genotyped --fasta target.fa --vcf merge.sites.vcf.gz S1.sort.markdup.bam # S2 S3... Sn
python3 LumpyFilter.py S1 results_genotyped

5. merge
svimmer --threads 10 allfile.list Chrname  > merge.vcf
graphtyper genotype_sv target.fa merge.vcf.gz --sams=bam.list --region=chr:start-end --threads=10
####根据染色体合并

6. filter
python3 FinalFilter.py sv.vcf.gz > sv.filter.vcf
vcftools --vcf sv.filter.vcf --max-missing 0.95 --recode --recode-INFO-all --out
